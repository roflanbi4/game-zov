func setup(target: Vector2, projectile_speed: float, projectile_damage: int, projectile_target_group: String) -> void:
	target_position = target
	speed = projectile_speed
	damage = projectile_damage
	target_group = projectile_target_group

func _process(delta: float) -> void:
	if not active:
		return
	var old_pos: Vector2 = global_position
	var dir: Vector2 = (target_position - global_position).normalized()
	var new_pos: Vector2 = global_position + dir * speed * delta
	var space: PhysicsDirectSpaceState2D = get_world_2d().direct_space_state
	if space != null:
		var query: PhysicsRayQueryParameters2D = PhysicsRayQueryParameters2D.create(old_pos, new_pos)
		if shooter != null and is_instance_valid(shooter):
			query.exclude = [shooter]
		var result: Dictionary = space.intersect_ray(query)
		if not result.is_empty():
			var collider: Object = result.get("collider")
			if collider is StaticBody2D:
				var sb: StaticBody2D = collider as StaticBody2D
				var hc: HealthComponent = sb.get_node_or_null("HealthComponent") as HealthComponent
				if not collider.has_meta("is_tall_wall") and shooter != null and is_instance_valid(shooter) and shooter.global_position.distance_to((collider as Node2D).global_position) <= 65.0:
					pass
				else:
					if hc != null:
						hc.take_damage(damage)
					active = false
					queue_free()
					return
	global_position = new_pos
	if global_position.distance_to(target_position) < 10.0:
		_hit_target()
		active = false

func _hit_target() -> void:
	var targets: Array = get_tree().get_nodes_in_group(target_group)
	targets.append_array(get_tree().get_nodes_in_group("explosives"))
	var final_damage: float = float(damage)
	for obj in targets:
		if is_instance_valid(obj) and global_position.distance_to(obj.global_position) < 24.0:
			if obj is Soldier:
				var facing_variant: Variant = obj.get("_last_facing")
				if facing_variant is Vector2:
					var facing: Vector2 = facing_variant as Vector2
					var to_attacker: Vector2 = (global_position - obj.global_position).normalized()
					var dot: float = facing.dot(to_attacker)
					if dot < -0.5:
						final_damage *= 1.5
					elif dot < 0.2:
						final_damage *= 1.25
			var health: HealthComponent = obj.get_node_or_null("HealthComponent")
			if health != null:
				if obj is VehicleUnit:
					(obj as VehicleUnit).take_damage(int(final_damage), is_anti_tank)
				elif obj.has_method("take_damage"):
					obj.take_damage(int(final_damage))
				else:
					health.take_damage(int(final_damage))
			elif obj.has_method("take_damage"):
				obj.take_damage(int(final_damage))
			if morale_penalty > 0:
				var morale: MoraleComponent = obj.get_node_or_null("MoraleComponent") as MoraleComponent
				if morale != null:
					morale.reduce_morale(float(morale_penalty))

func _pick_free_spot(rng: RandomNumberGenerator, placed: Array[Vector2], min_dist: float) -> Vector2:
	for attempt in range(40):
		var x: float = float(rng.randi_range(450, 2150))
		var y: float = float(rng.randi_range(180, 1260))
		if absf(y - _CFG.MAP_HEIGHT * 0.5) < 110.0:
			continue
		var candidate := Vector2(x, y)
		var ok: bool = true
		for p in placed:
			if p.distance_to(candidate) < min_dist:
				ok = false
				break
		if ok:
			return candidate
	return Vector2(float(rng.randi_range(450, 2150)), _CFG.MAP_HEIGHT * 0.5)

	return origin

	target_position = global_position
	velocity = Vector2.ZERO

func set_attack_move(enabled: bool) -> void:
	attack_move = enabled

func _physics_process(_delta: float) -> void:
	if disabled:
		velocity = Vector2.ZERO
		move_and_slide()
		return
	if target_position == Vector2.ZERO:
		velocity = Vector2.ZERO
		move_and_slide()
		return

	var to_target: Vector2 = target_position - global_position
	if drone_type == DroneType.FPV_KAMIKAZE:
		if _check_detonation():
			_explode()
			return

	if to_target.length() < 8.0:
		velocity = Vector2.ZERO
		move_and_slide()
		set_physics_process(false)
		return

	velocity = to_target.normalized() * move_speed
	move_and_slide()

func _check_detonation() -> bool:
	for group in _target_groups():
		for node in get_tree().get_nodes_in_group(group):
			if is_instance_valid(node) and node != self:
				if global_position.distance_to(node.global_position) <= 20.0:
					return true
	return false

func _target_groups() -> Array[String]:
	if is_enemy:
		return ["player_units", "player_vehicles"]
	return ["enemy_units", "enemy_vehicles"]

func take_damage(amount: int) -> void:
	current_hp -= amount
	if current_hp <= 0:
		current_hp = 0
		if drone_type == DroneType.FPV_KAMIKAZE and not _exploded:
			_explode()
		else:
			queue_free()

func _explode() -> void:
	if _exploded:
		return
	_exploded = true
	for group in _target_groups():
		for node in get_tree().get_nodes_in_group(group):
			if not is_instance_valid(node) or node == self:
				continue
			if global_position.distance_to(node.global_position) > explosion_radius:
				continue
			var dmg: int = vehicle_damage if node.is_in_group("player_vehicles") or node.is_in_group("enemy_vehicles") else explosion_damage
			if node.has_method("take_damage"):
				node.take_damage(dmg)
			else:
				var health: HealthComponent = node.get_node_or_null("HealthComponent") as HealthComponent
				if health != null:
					health.take_damage(dmg)
	queue_free()
